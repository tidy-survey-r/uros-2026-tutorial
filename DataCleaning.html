ess_der_all <-
  ess_in |>
  select(
    psu,
    cntry,
    stratum,
    anweight,
    happy,
    wrhpp,
    nwspol,
    volunfp,
    netusoft,
    netustm,
    agegroup,
    pdwrk:dngna
  ) |>
  mutate(
    # across(c(stratum, psu), ~as.character(.x) |> as.factor()),
    across(
      pdwrk:dngoth,
      ~ if_else(dngdk == 1 | dngref == 1 | dngna == 1, NA, .x),
      .names = "{.col}_rec"
    ),
    Retired = rtrd_rec == 1,
    wt1 = 1,
    Happy_Freq = if_else(wrhpp %in% 7:9, NA, wrhpp),
    Happy = if_else(happy %in% c(77, 88, 99), NA, happy),
    News = if_else(nwspol %in% c(7777, 8888, 9999), NA, nwspol),
    Country = recode_values(
      cntry,
      from = country_coding$Code,
      to = country_coding$Name
    ),
    Volunteer_12 = if_else(volunfp %in% 7:9, NA, volunfp),
    Volunteer_01 = case_when(
      volunfp == 1 ~ 1,
      volunfp == 2 ~ 0,
      TRUE ~ NA
    ),
    Volunteer_Factor = case_when(
      volunfp == 1 ~ "Yes",
      volunfp == 2 ~ "No",
      TRUE ~ NA
    ) |>
      factor(levels = c("Yes", "No")),
    NetUsage = case_when(
      netusoft %in% 1:3 ~ 0,
      netustm %in% c(6666, 7777, 8888, 9999) ~ NA,
      TRUE ~ netustm
    ),
    NetFreq = recode_values(
      netusoft,
      1 ~ "Never",
      2 ~ "Only occasionally",
      3 ~ "A few times a week",
      4 ~ "Most days",
      5 ~ "Every day"
    ) |>
      fct_reorder(netusoft),
    AgeGroup = recode_values(
      agegroup,
      1 ~ '15-24',
      2 ~ '25-34',
      3 ~ '35-44',
      4 ~ '45-54',
      5 ~ '55-64',
      6 ~ '65-74',
      7 ~ '75+'
    ) |>
      fct_reorder(agegroup)
  )
